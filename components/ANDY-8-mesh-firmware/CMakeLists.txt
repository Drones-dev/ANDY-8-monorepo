cmake_minimum_required(VERSION 3.16)
project(andy8_mesh_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)
add_library(andy_mesh_core firmware/mesh-node/PacketProtocol.cpp firmware/mesh-node/NeighborTable.cpp firmware/mesh-node/HelloManager.cpp)
target_include_directories(andy_mesh_core PUBLIC firmware/mesh-node)
add_library(andy_sim_radio simulation/SimulatedRadio.cpp)
target_include_directories(andy_sim_radio PUBLIC simulation firmware/mesh-node)
target_link_libraries(andy_sim_radio PUBLIC andy_mesh_core)
add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE andy_mesh_core)
add_executable(simulation_tests tests/simulation_tests.cpp)
target_link_libraries(simulation_tests PRIVATE andy_sim_radio)
enable_testing()
add_test(NAME protocol COMMAND protocol_tests)
add_test(NAME simulated_radio COMMAND simulation_tests)
if(MAVLINK_INCLUDE_DIR)
  add_executable(andy_mesh_sim simulation/mesh_sim.cpp firmware/mesh-node/MavlinkTelemetryDecoder.cpp)
  target_include_directories(andy_mesh_sim PRIVATE simulation firmware/mesh-node ${MAVLINK_INCLUDE_DIR})
  target_compile_definitions(andy_mesh_sim PRIVATE ANDY_TELEMETRY_SOURCE=1 ANDY_MAVLINK_RX_PIN=0 ANDY_MAVLINK_TX_PIN=0 ANDY_MAVLINK_BAUD=57600)
  target_link_libraries(andy_mesh_sim PRIVATE andy_sim_radio)
endif()
